CMAKE_MINIMUM_REQUIRED(VERSION 3.8.2)

PROJECT(flexlayout.example)

SET(CMAKE_CXX_STANDARD 17)
SET(CMAKE_C_STANDARD 99)

# Find required packages
FIND_PACKAGE(PkgConfig REQUIRED)
PKG_CHECK_MODULES(DALICORE REQUIRED dali2-core)
PKG_CHECK_MODULES(DALIADAPTOR REQUIRED dali2-adaptor)
PKG_CHECK_MODULES(DALI_UI_FOUNDATION REQUIRED dali2-ui-foundation)
PKG_CHECK_MODULES(DALI_UI_COMPONENTS REQUIRED dali2-ui-components)

# Add executables
SET(EXE_NAME flexlayout.example)
ADD_EXECUTABLE(${EXE_NAME} flexlayout-example.cpp)

SET(EXE_JUSTIFY_ALIGN_NAME flexlayout-justify-align.example)
ADD_EXECUTABLE(${EXE_JUSTIFY_ALIGN_NAME} flexlayout-justify-align-example.cpp)

SET(EXE_WRAP_GROW_NAME flexlayout-wrap-grow.example)
ADD_EXECUTABLE(${EXE_WRAP_GROW_NAME} flexlayout-wrap-grow-example.cpp)

SET(EXE_MARGIN_PADDING_NAME flexlayout-margin-padding.example)
ADD_EXECUTABLE(${EXE_MARGIN_PADDING_NAME} flexlayout-margin-padding-example.cpp)

SET(EXE_DIRECTION_REVERSE_NAME flexlayout-direction-reverse.example)
ADD_EXECUTABLE(${EXE_DIRECTION_REVERSE_NAME} flexlayout-direction-reverse-example.cpp)

SET(EXE_JUSTIFY_CONTENT_NAME flexlayout-justify-content.example)
ADD_EXECUTABLE(${EXE_JUSTIFY_CONTENT_NAME} flexlayout-justify-content-example.cpp)

SET(EXE_SHRINK_NAME flexlayout-shrink.example)
ADD_EXECUTABLE(${EXE_SHRINK_NAME} flexlayout-shrink-example.cpp)

SET(EXE_WRAPCONTENT_GROW_NAME flexlayout-wrapcontent-grow.example)
ADD_EXECUTABLE(${EXE_WRAPCONTENT_GROW_NAME} flexlayout-wrapcontent-grow-example.cpp)

SET(EXE_LAYOUT_DIRECTION_NAME flexlayout-layout-direction.example)
ADD_EXECUTABLE(${EXE_LAYOUT_DIRECTION_NAME} flexlayout-layout-direction-example.cpp)

# Set output directory
SET_TARGET_PROPERTIES(${EXE_NAME} ${EXE_JUSTIFY_ALIGN_NAME} ${EXE_WRAP_GROW_NAME} ${EXE_MARGIN_PADDING_NAME} ${EXE_DIRECTION_REVERSE_NAME} ${EXE_JUSTIFY_CONTENT_NAME} ${EXE_SHRINK_NAME} ${EXE_WRAPCONTENT_GROW_NAME} ${EXE_LAYOUT_DIRECTION_NAME} PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")

INSTALL(TARGETS ${EXE_NAME} ${EXE_JUSTIFY_ALIGN_NAME} ${EXE_WRAP_GROW_NAME} ${EXE_MARGIN_PADDING_NAME} ${EXE_DIRECTION_REVERSE_NAME} ${EXE_JUSTIFY_CONTENT_NAME} ${EXE_SHRINK_NAME} ${EXE_WRAPCONTENT_GROW_NAME} ${EXE_LAYOUT_DIRECTION_NAME} RUNTIME DESTINATION)

# Build Options
SET(DALI_UI_TEST_APP_CFLAGS -fvisibility=hidden -DHIDE_DALI_INTERNALS)

if(CMAKE_BUILD_TYPE MATCHES Debug)
  MESSAGE(STATUS "CMAKE_BUILD_TYPE: " Debug)
  SET(DALI_UI_TEST_APP_CFLAGS ${DALI_UI_TEST_APP_CFLAGS} -g -O0 -DDEBUG_ENABLED)
ELSE()
  MESSAGE(STATUS "CMAKE_BUILD_TYPE: " Release)
  SET(DALI_UI_TEST_APP_CFLAGS ${DALI_UI_TEST_APP_CFLAGS} -Os)
ENDIF()

# Add -rdynamic to export main symbol, so dali-loader could run the main.
SET(DALI_UI_TEST_APP_CFLAGS ${DALI_UI_TEST_APP_CFLAGS} -Werror -Wall -fPIE -rdynamic)

# Remove some warnings that are not useful for our test apps.
IF( WIN32 )
  # TODO: Add Windows-specific compiler flags if needed.
ELSE()
  INCLUDE(CheckCXXCompilerFlag)

  CHECK_CXX_COMPILER_FLAG( -Wno-unused-const-variable HAVE_NO_UNUSED_CONST_VARIABLE )
  IF( HAVE_NO_UNUSED_CONST_VARIABLE )
    SET(DALI_UI_TEST_APP_CFLAGS ${DALI_UI_TEST_APP_CFLAGS} -Wno-unused-const-variable)
  ENDIF()

  CHECK_CXX_COMPILER_FLAG( -Wno-unused-command-line-argument HAVE_NO_UNUSED_COMMAND_LINE_ARGUMENT )
  IF( HAVE_NO_UNUSED_COMMAND_LINE_ARGUMENT )
    SET(DALI_UI_TEST_APP_CFLAGS ${DALI_UI_TEST_APP_CFLAGS} -Wno-unused-command-line-argument)
  ENDIF()
ENDIF()
