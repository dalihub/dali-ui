CMAKE_MINIMUM_REQUIRED(VERSION 3.8.2)

PROJECT(flexlayout.example)

SET(CMAKE_CXX_STANDARD 17)
SET(CMAKE_C_STANDARD 99)

# Find required packages
FIND_PACKAGE(PkgConfig REQUIRED)
PKG_CHECK_MODULES(DALICORE REQUIRED dali2-core)
PKG_CHECK_MODULES(DALIADAPTOR REQUIRED dali2-adaptor)
PKG_CHECK_MODULES(DALI_UI_FOUNDATION REQUIRED dali2-ui-foundation)
PKG_CHECK_MODULES(DALI_UI_COMPONENTS REQUIRED dali2-ui-components)

# Add executables
SET(EXE_NAME flexlayout.example)
ADD_EXECUTABLE(${EXE_NAME} flexlayout-example.cpp)

SET(EXE_JUSTIFY_ALIGN_NAME flexlayout-justify-align.example)
ADD_EXECUTABLE(${EXE_JUSTIFY_ALIGN_NAME} flexlayout-justify-align-example.cpp)

SET(EXE_WRAP_GROW_NAME flexlayout-wrap-grow.example)
ADD_EXECUTABLE(${EXE_WRAP_GROW_NAME} flexlayout-wrap-grow-example.cpp)

SET(EXE_MARGIN_PADDING_NAME flexlayout-margin-padding.example)
ADD_EXECUTABLE(${EXE_MARGIN_PADDING_NAME} flexlayout-margin-padding-example.cpp)

SET(EXE_DIRECTION_REVERSE_NAME flexlayout-direction-reverse.example)
ADD_EXECUTABLE(${EXE_DIRECTION_REVERSE_NAME} flexlayout-direction-reverse-example.cpp)

SET(EXE_JUSTIFY_CONTENT_NAME flexlayout-justify-content.example)
ADD_EXECUTABLE(${EXE_JUSTIFY_CONTENT_NAME} flexlayout-justify-content-example.cpp)

SET(EXE_SHRINK_NAME flexlayout-shrink.example)
ADD_EXECUTABLE(${EXE_SHRINK_NAME} flexlayout-shrink-example.cpp)

# Set output directory
SET_TARGET_PROPERTIES(${EXE_NAME} ${EXE_JUSTIFY_ALIGN_NAME} ${EXE_WRAP_GROW_NAME} ${EXE_MARGIN_PADDING_NAME} ${EXE_DIRECTION_REVERSE_NAME} ${EXE_JUSTIFY_CONTENT_NAME} ${EXE_SHRINK_NAME} PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")

INSTALL(TARGETS ${EXE_NAME} ${EXE_JUSTIFY_ALIGN_NAME} ${EXE_WRAP_GROW_NAME} ${EXE_MARGIN_PADDING_NAME} ${EXE_DIRECTION_REVERSE_NAME} ${EXE_JUSTIFY_CONTENT_NAME} ${EXE_SHRINK_NAME} RUNTIME DESTINATION)

# Link libraries
TARGET_LINK_LIBRARIES(${EXE_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_JUSTIFY_ALIGN_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_WRAP_GROW_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_MARGIN_PADDING_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_DIRECTION_REVERSE_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_JUSTIFY_CONTENT_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_SHRINK_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)

# Add link directories
TARGET_LINK_DIRECTORIES(${EXE_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_JUSTIFY_ALIGN_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_WRAP_GROW_NAME} PRIVATE

TARGET_LINK_DIRECTORIES(${EXE_SHRINK_NAME} PRIVATE
