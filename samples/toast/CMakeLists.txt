CMAKE_MINIMUM_REQUIRED(VERSION 3.8.2)
PROJECT(toast.example)
IF(NOT COMMAND DALI_ADD_SAMPLE)
  IF(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../common.cmake")
    INCLUDE("${CMAKE_CURRENT_LIST_DIR}/../common.cmake")
  ELSE()
    # A standalone GBS export does not contain the parent samples directory.
    INCLUDE(GNUInstallDirs)
    FIND_PACKAGE(PkgConfig REQUIRED)
    PKG_CHECK_MODULES(DALICORE REQUIRED IMPORTED_TARGET dali2-core)
    PKG_CHECK_MODULES(DALIADAPTOR REQUIRED IMPORTED_TARGET dali2-adaptor)
    PKG_CHECK_MODULES(DALI_UI_FOUNDATION REQUIRED IMPORTED_TARGET dali2-ui-foundation)
    PKG_CHECK_MODULES(DALI_UI_COMPONENTS REQUIRED IMPORTED_TARGET dali2-ui-components)

    FUNCTION(DALI_ADD_SAMPLE EXE_NAME)
      ADD_EXECUTABLE(${EXE_NAME} ${ARGN})
      SET_TARGET_PROPERTIES(${EXE_NAME} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin"
      )
      IF(CMAKE_BUILD_TYPE MATCHES Debug)
        TARGET_COMPILE_DEFINITIONS(${EXE_NAME} PRIVATE DEBUG_ENABLED)
      ENDIF()
      TARGET_LINK_LIBRARIES(${EXE_NAME} PRIVATE
        PkgConfig::DALI_UI_COMPONENTS
        PkgConfig::DALI_UI_FOUNDATION
        PkgConfig::DALIADAPTOR
        PkgConfig::DALICORE
      )
      TARGET_COMPILE_DEFINITIONS(${EXE_NAME} PRIVATE HIDE_DALI_INTERNALS)
      TARGET_COMPILE_OPTIONS(${EXE_NAME} PRIVATE
        -fvisibility=hidden
        -Wall
        -fPIE
        -rdynamic
      )
      INSTALL(TARGETS ${EXE_NAME} RUNTIME DESTINATION bin)
    ENDFUNCTION()
  ENDIF()
ENDIF()
DALI_ADD_SAMPLE(toast.example toast-example.cpp)
