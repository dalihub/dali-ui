CMAKE_MINIMUM_REQUIRED(VERSION 3.8.2)

PROJECT(page-scroller.example)

SET(CMAKE_CXX_STANDARD 17)
SET(CMAKE_C_STANDARD 99)

IF(NOT COMMAND DALI_ADD_SAMPLE)
  INCLUDE("${CMAKE_CURRENT_LIST_DIR}/../common.cmake")
ENDIF()

SET(PAGE_SCROLLER_TARGETS
  page-scroll-view-example
  page-scroll-view-dynamic-example
)

FOREACH(TARGET_NAME ${PAGE_SCROLLER_TARGETS})
  DALI_ADD_SAMPLE(${TARGET_NAME} ${TARGET_NAME}.cpp)
  TARGET_INCLUDE_DIRECTORIES(${TARGET_NAME} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../../
  )
ENDFOREACH()
