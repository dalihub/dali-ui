CMAKE_MINIMUM_REQUIRED(VERSION 3.8.2)

PROJECT(absolutelayout.example)

SET(CMAKE_CXX_STANDARD 17)
SET(CMAKE_C_STANDARD 99)

# Find required packages
FIND_PACKAGE(PkgConfig REQUIRED)
PKG_CHECK_MODULES(DALICORE REQUIRED dali2-core)
PKG_CHECK_MODULES(DALIADAPTOR REQUIRED dali2-adaptor)
PKG_CHECK_MODULES(DALI_UI_FOUNDATION REQUIRED dali2-ui-foundation)
PKG_CHECK_MODULES(DALI_UI_COMPONENTS REQUIRED dali2-ui-components)

# Add executables
SET(EXE_NAME absolutelayout.example)
ADD_EXECUTABLE(${EXE_NAME} absolutelayout-example.cpp)

SET(EXE_PROPORTIONAL_NAME absolutelayout-proportional.example)
ADD_EXECUTABLE(${EXE_PROPORTIONAL_NAME} absolutelayout-proportional-example.cpp)

SET(EXE_OVERLAP_NAME absolutelayout-overlap.example)
ADD_EXECUTABLE(${EXE_OVERLAP_NAME} absolutelayout-overlap-example.cpp)

SET(EXE_MARGIN_PADDING_NAME absolutelayout-margin-padding.example)
ADD_EXECUTABLE(${EXE_MARGIN_PADDING_NAME} absolutelayout-margin-padding-example.cpp)

SET(EXE_WRAPCONTENT_PROPORTIONAL_NAME absolutelayout-wrapcontent-proportional.example)
ADD_EXECUTABLE(${EXE_WRAPCONTENT_PROPORTIONAL_NAME} absolutelayout-wrapcontent-proportional-example.cpp)

SET(EXE_PER_AXIS_PROPORTIONAL_NAME absolutelayout-per-axis-proportional.example)
ADD_EXECUTABLE(${EXE_PER_AXIS_PROPORTIONAL_NAME} absolutelayout-per-axis-proportional-example.cpp)

SET(EXE_LAYOUT_DIRECTION_NAME absolutelayout-layout-direction.example)
ADD_EXECUTABLE(${EXE_LAYOUT_DIRECTION_NAME} absolutelayout-layout-direction-example.cpp)

# Set output directory
SET_TARGET_PROPERTIES(${EXE_NAME} ${EXE_PROPORTIONAL_NAME} ${EXE_OVERLAP_NAME} ${EXE_MARGIN_PADDING_NAME} ${EXE_WRAPCONTENT_PROPORTIONAL_NAME} ${EXE_PER_AXIS_PROPORTIONAL_NAME} ${EXE_LAYOUT_DIRECTION_NAME} PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")

INSTALL(TARGETS ${EXE_NAME} ${EXE_PROPORTIONAL_NAME} ${EXE_OVERLAP_NAME} ${EXE_MARGIN_PADDING_NAME} ${EXE_WRAPCONTENT_PROPORTIONAL_NAME} ${EXE_PER_AXIS_PROPORTIONAL_NAME} ${EXE_LAYOUT_DIRECTION_NAME} RUNTIME DESTINATION)

# Build Options
SET(DALI_UI_TEST_APP_CFLAGS -fvisibility=hidden -DHIDE_DALI_INTERNALS)

if(CMAKE_BUILD_TYPE MATCHES Debug)
  MESSAGE(STATUS "CMAKE_BUILD_TYPE: " Debug)
  SET(DALI_UI_TEST_APP_CFLAGS ${DALI_UI_TEST_APP_CFLAGS} -g -O0 -DDEBUG_ENABLED)
ELSE()
  MESSAGE(STATUS "CMAKE_BUILD_TYPE: " Release)
  SET(DALI_UI_TEST_APP_CFLAGS ${DALI_UI_TEST_APP_CFLAGS} -Os)
ENDIF()

# Add -rdynamic to export main symbol, so dali-loader could run the main.
SET(DALI_UI_TEST_APP_CFLAGS ${DALI_UI_TEST_APP_CFLAGS} -Werror -Wall -fPIE -rdynamic)

# Link libraries
TARGET_LINK_LIBRARIES(${EXE_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_PROPORTIONAL_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_OVERLAP_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_MARGIN_PADDING_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_WRAPCONTENT_PROPORTIONAL_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_PER_AXIS_PROPORTIONAL_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_LAYOUT_DIRECTION_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)

# Add link directories
TARGET_LINK_DIRECTORIES(${EXE_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_PROPORTIONAL_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_OVERLAP_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_MARGIN_PADDING_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_WRAPCONTENT_PROPORTIONAL_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_PER_AXIS_PROPORTIONAL_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_LAYOUT_DIRECTION_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)

TARGET_INCLUDE_DIRECTORIES(${EXE_NAME} PRIVATE

  ${DALICORE_INCLUDE_DIRS}

TARGET_INCLUDE_DIRECTORIES(${EXE_LAYOUT_DIRECTION_NAME} PRIVATE

  ${DALI_UI_COMPONENTS_INCLUDE_DIRS}
)
