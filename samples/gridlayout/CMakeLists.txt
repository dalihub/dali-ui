CMAKE_MINIMUM_REQUIRED(VERSION 3.8.2)

PROJECT(gridlayout.example)

SET(CMAKE_CXX_STANDARD 17)
SET(CMAKE_C_STANDARD 99)

# Find required packages
FIND_PACKAGE(PkgConfig REQUIRED)
PKG_CHECK_MODULES(DALICORE REQUIRED dali2-core)
PKG_CHECK_MODULES(DALIADAPTOR REQUIRED dali2-adaptor)
PKG_CHECK_MODULES(DALI_UI_FOUNDATION REQUIRED dali2-ui-foundation)
PKG_CHECK_MODULES(DALI_UI_COMPONENTS REQUIRED dali2-ui-components)

# Add executables
SET(EXE_NAME gridlayout.example)
ADD_EXECUTABLE(${EXE_NAME} gridlayout-example.cpp)

SET(EXE_STAR_AUTO_NAME gridlayout-star-auto.example)
ADD_EXECUTABLE(${EXE_STAR_AUTO_NAME} gridlayout-star-auto-example.cpp)

SET(EXE_SPAN_NAME gridlayout-span.example)
ADD_EXECUTABLE(${EXE_SPAN_NAME} gridlayout-span-example.cpp)

SET(EXE_MARGIN_PADDING_NAME gridlayout-margin-padding.example)
ADD_EXECUTABLE(${EXE_MARGIN_PADDING_NAME} gridlayout-margin-padding-example.cpp)

SET(EXE_AUTO_NAME gridlayout-auto.example)
ADD_EXECUTABLE(${EXE_AUTO_NAME} gridlayout-auto-example.cpp)

SET(EXE_WRAPCONTENT_STAR_NAME gridlayout-wrapcontent-star.example)
ADD_EXECUTABLE(${EXE_WRAPCONTENT_STAR_NAME} gridlayout-wrapcontent-star-example.cpp)

SET(EXE_LAYOUT_DIRECTION_NAME gridlayout-layout-direction.example)
ADD_EXECUTABLE(${EXE_LAYOUT_DIRECTION_NAME} gridlayout-layout-direction-example.cpp)

# Set output directory
SET_TARGET_PROPERTIES(${EXE_NAME} ${EXE_STAR_AUTO_NAME} ${EXE_SPAN_NAME} ${EXE_MARGIN_PADDING_NAME} ${EXE_AUTO_NAME} ${EXE_WRAPCONTENT_STAR_NAME} ${EXE_LAYOUT_DIRECTION_NAME} PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")

INSTALL(TARGETS ${EXE_NAME} ${EXE_STAR_AUTO_NAME} ${EXE_SPAN_NAME} ${EXE_MARGIN_PADDING_NAME} ${EXE_AUTO_NAME} ${EXE_WRAPCONTENT_STAR_NAME} ${EXE_LAYOUT_DIRECTION_NAME} RUNTIME DESTINATION)

# Build Options
SET(DALI_UI_TEST_APP_CFLAGS -fvisibility=hidden -DHIDE_DALI_INTERNALS)

if(CMAKE_BUILD_TYPE MATCHES Debug)
  MESSAGE(STATUS "CMAKE_BUILD_TYPE: " Debug)
  SET(DALI_UI_TEST_APP_CFLAGS ${DALI_UI_TEST_APP_CFLAGS} -g -O0 -DDEBUG_ENABLED)
ELSE()
  MESSAGE(STATUS "CMAKE_BUILD_TYPE: " Release)
  SET(DALI_UI_TEST_APP_CFLAGS ${DALI_UI_TEST_APP_CFLAGS} -Os)
ENDIF()

# Add -rdynamic to export main symbol, so dali-loader could run the main.
SET(DALI_UI_TEST_APP_CFLAGS ${DALI_UI_TEST_APP_CFLAGS} -Werror -Wall -fPIE -rdynamic)

# Link libraries
TARGET_LINK_LIBRARIES(${EXE_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_STAR_AUTO_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_SPAN_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_MARGIN_PADDING_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_AUTO_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_WRAPCONTENT_STAR_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_LAYOUT_DIRECTION_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)

# Add link directories
TARGET_LINK_DIRECTORIES(${EXE_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_STAR_AUTO_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_SPAN_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)

  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_WRAPCONTENT_STAR_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_LAYOUT_DIRECTION_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
