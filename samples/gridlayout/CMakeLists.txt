CMAKE_MINIMUM_REQUIRED(VERSION 3.8.2)

PROJECT(gridlayout.example)

SET(CMAKE_CXX_STANDARD 17)
SET(CMAKE_C_STANDARD 99)

# Find required packages
FIND_PACKAGE(PkgConfig REQUIRED)
PKG_CHECK_MODULES(DALICORE REQUIRED dali2-core)
PKG_CHECK_MODULES(DALIADAPTOR REQUIRED dali2-adaptor)
PKG_CHECK_MODULES(DALITOOLKIT REQUIRED dali2-toolkit)
PKG_CHECK_MODULES(DALI_UI_FOUNDATION REQUIRED dali2-ui-foundation)
PKG_CHECK_MODULES(DALI_UI_COMPONENTS REQUIRED dali2-ui-components)

# Add executables
SET(EXE_NAME gridlayout.example)
ADD_EXECUTABLE(${EXE_NAME} gridlayout-example.cpp)

SET(EXE_STAR_AUTO_NAME gridlayout-star-auto.example)
ADD_EXECUTABLE(${EXE_STAR_AUTO_NAME} gridlayout-star-auto-example.cpp)

SET(EXE_SPAN_NAME gridlayout-span.example)
ADD_EXECUTABLE(${EXE_SPAN_NAME} gridlayout-span-example.cpp)

SET(EXE_MARGIN_PADDING_NAME gridlayout-margin-padding.example)
ADD_EXECUTABLE(${EXE_MARGIN_PADDING_NAME} gridlayout-margin-padding-example.cpp)

SET(EXE_AUTO_NAME gridlayout-auto.example)
ADD_EXECUTABLE(${EXE_AUTO_NAME} gridlayout-auto-example.cpp)

# Set output directory
SET_TARGET_PROPERTIES(${EXE_NAME} ${EXE_STAR_AUTO_NAME} ${EXE_SPAN_NAME} ${EXE_MARGIN_PADDING_NAME} ${EXE_AUTO_NAME} PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")

INSTALL(TARGETS ${EXE_NAME} ${EXE_STAR_AUTO_NAME} ${EXE_SPAN_NAME} ${EXE_MARGIN_PADDING_NAME} ${EXE_AUTO_NAME} RUNTIME DESTINATION)

# Link libraries
TARGET_LINK_LIBRARIES(${EXE_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALITOOLKIT_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_STAR_AUTO_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALITOOLKIT_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_SPAN_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALITOOLKIT_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_MARGIN_PADDING_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALITOOLKIT_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_AUTO_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALITOOLKIT_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)

# Add link directories
TARGET_LINK_DIRECTORIES(${EXE_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_STAR_AUTO_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_SPAN_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_MARGIN_PADDING_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_AUTO_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)

# Include directories
TARGET_INCLUDE_DIRECTORIES(${EXE_NAME} PRIVATE
  ${DALICORE_INCLUDE_DIRS}
  ${DALIADAPTOR_INCLUDE_DIRS}
  ${DALITOOLKIT_INCLUDE_DIRS}

TARGET_INCLUDE_DIRECTORIES(${EXE_MARGIN_PADDING_NAME} PRIVATE
  ${DALICORE_INCLUDE_DIRS}
  ${DALIADAPTOR_INCLUDE_DIRS}
  ${DALITOOLKIT_INCLUDE_DIRS}
  ${DALI_UI_FOUNDATION_INCLUDE_DIRS}
  ${DALI_UI_COMPONENTS_INCLUDE_DIRS}
  ${DALI_UI_COMPONENTS_INCLUDE_DIRS}
)
TARGET_INCLUDE_DIRECTORIES(${EXE_AUTO_NAME} PRIVATE
  ${DALICORE_INCLUDE_DIRS}
  ${DALIADAPTOR_INCLUDE_DIRS}
  ${DALITOOLKIT_INCLUDE_DIRS}
  ${DALI_UI_FOUNDATION_INCLUDE_DIRS}
  ${DALI_UI_COMPONENTS_INCLUDE_DIRS}
  ${DALI_UI_COMPONENTS_INCLUDE_DIRS}
)
