CMAKE_MINIMUM_REQUIRED(VERSION 3.8.2)

PROJECT(layout-transition.example)

SET(CMAKE_CXX_STANDARD 17)
SET(CMAKE_C_STANDARD 99)

# Find required packages
FIND_PACKAGE(PkgConfig REQUIRED)
PKG_CHECK_MODULES(DALICORE REQUIRED dali2-core)
PKG_CHECK_MODULES(DALIADAPTOR REQUIRED dali2-adaptor)
PKG_CHECK_MODULES(DALI_UI_FOUNDATION REQUIRED dali2-ui-foundation)
PKG_CHECK_MODULES(DALI_UI_COMPONENTS REQUIRED dali2-ui-components)

SET(EXE_SPEC_NAME layout-transition-spec.example)
ADD_EXECUTABLE(${EXE_SPEC_NAME} layout-transition-spec-example.cpp)

SET(EXE_ANIMATOR_NAME layout-transition-animator.example)
ADD_EXECUTABLE(${EXE_ANIMATOR_NAME} layout-transition-animator-example.cpp)

SET(EXE_REORDER_NAME layout-transition-reorder.example)
ADD_EXECUTABLE(${EXE_REORDER_NAME} layout-transition-reorder-example.cpp)

SET(EXE_SUBTREE_NAME layout-transition-subtree.example)
ADD_EXECUTABLE(${EXE_SUBTREE_NAME} layout-transition-subtree-example.cpp)

SET(EXE_GRID_REORDER_NAME layout-transition-grid-reorder.example)
ADD_EXECUTABLE(${EXE_GRID_REORDER_NAME} layout-transition-grid-reorder-example.cpp)

SET_TARGET_PROPERTIES(${EXE_SPEC_NAME} ${EXE_ANIMATOR_NAME} ${EXE_REORDER_NAME} ${EXE_SUBTREE_NAME} ${EXE_GRID_REORDER_NAME} PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")

INSTALL(TARGETS ${EXE_SPEC_NAME} ${EXE_ANIMATOR_NAME} ${EXE_REORDER_NAME} ${EXE_SUBTREE_NAME} ${EXE_GRID_REORDER_NAME} RUNTIME DESTINATION)

# Build options
SET(DALI_UI_TEST_APP_CFLAGS -fvisibility=hidden -DHIDE_DALI_INTERNALS)

if(CMAKE_BUILD_TYPE MATCHES Debug)
  MESSAGE(STATUS "CMAKE_BUILD_TYPE: " Debug)
  SET(DALI_UI_TEST_APP_CFLAGS ${DALI_UI_TEST_APP_CFLAGS} -g -O0 -DDEBUG_ENABLED)
ELSE()
  MESSAGE(STATUS "CMAKE_BUILD_TYPE: " Release)
  SET(DALI_UI_TEST_APP_CFLAGS ${DALI_UI_TEST_APP_CFLAGS} -Os)
ENDIF()

# Add -rdynamic to export main symbol, so dali-loader could run the main.
SET(DALI_UI_TEST_APP_CFLAGS ${DALI_UI_TEST_APP_CFLAGS} -Werror -Wall -fPIE -rdynamic)

# Remove some warnings that are not useful for our test apps.
IF( WIN32 )
  # TODO: Add Windows-specific compiler flags if needed.
ELSE()
  INCLUDE(CheckCXXCompilerFlag)

  CHECK_CXX_COMPILER_FLAG( -Wno-unused-const-variable HAVE_NO_UNUSED_CONST_VARIABLE )
  IF( HAVE_NO_UNUSED_CONST_VARIABLE )
    SET(DALI_UI_TEST_APP_CFLAGS ${DALI_UI_TEST_APP_CFLAGS} -Wno-unused-const-variable)
  ENDIF()

  CHECK_CXX_COMPILER_FLAG( -Wno-unused-command-line-argument HAVE_NO_UNUSED_COMMAND_LINE_ARGUMENT )
  IF( HAVE_NO_UNUSED_COMMAND_LINE_ARGUMENT )
    SET(DALI_UI_TEST_APP_CFLAGS ${DALI_UI_TEST_APP_CFLAGS} -Wno-unused-command-line-argument)
  ENDIF()
ENDIF()

TARGET_LINK_LIBRARIES(${EXE_SPEC_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_ANIMATOR_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_REORDER_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_SUBTREE_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)
TARGET_LINK_LIBRARIES(${EXE_GRID_REORDER_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)

TARGET_LINK_DIRECTORIES(${EXE_SPEC_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_ANIMATOR_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_REORDER_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_SUBTREE_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
TARGET_LINK_DIRECTORIES(${EXE_GRID_REORDER_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)

TARGET_INCLUDE_DIRECTORIES(${EXE_SPEC_NAME} PRIVATE
  ${DALICORE_INCLUDE_DIRS}
  ${DALIADAPTOR_INCLUDE_DIRS}
  ${DALI_UI_FOUNDATION_INCLUDE_DIRS}
  ${DALI_UI_COMPONENTS_INCLUDE_DIRS}
)
TARGET_INCLUDE_DIRECTORIES(${EXE_ANIMATOR_NAME} PRIVATE
  ${DALICORE_INCLUDE_DIRS}
  ${DALIADAPTOR_INCLUDE_DIRS}
  ${DALI_UI_FOUNDATION_INCLUDE_DIRS}
  ${DALI_UI_COMPONENTS_INCLUDE_DIRS}
)
TARGET_INCLUDE_DIRECTORIES(${EXE_REORDER_NAME} PRIVATE
  ${DALICORE_INCLUDE_DIRS}
  ${DALIADAPTOR_INCLUDE_DIRS}
  ${DALI_UI_FOUNDATION_INCLUDE_DIRS}
  ${DALI_UI_COMPONENTS_INCLUDE_DIRS}
)
TARGET_INCLUDE_DIRECTORIES(${EXE_SUBTREE_NAME} PRIVATE
  ${DALICORE_INCLUDE_DIRS}
  ${DALIADAPTOR_INCLUDE_DIRS}
  ${DALI_UI_FOUNDATION_INCLUDE_DIRS}
  ${DALI_UI_COMPONENTS_INCLUDE_DIRS}
)
TARGET_INCLUDE_DIRECTORIES(${EXE_GRID_REORDER_NAME} PRIVATE
  ${DALICORE_INCLUDE_DIRS}
  ${DALIADAPTOR_INCLUDE_DIRS}
  ${DALI_UI_FOUNDATION_INCLUDE_DIRS}
  ${DALI_UI_COMPONENTS_INCLUDE_DIRS}
)

TARGET_COMPILE_OPTIONS(${EXE_SPEC_NAME} PRIVATE
  ${DALICORE_CFLAGS_OTHER}
  ${DALIADAPTOR_CFLAGS_OTHER}
  ${DALI_UI_FOUNDATION_CFLAGS_OTHER}
  ${DALI_UI_COMPONENTS_CFLAGS_OTHER}
  ${DALI_UI_TEST_APP_CFLAGS}
)
TARGET_COMPILE_OPTIONS(${EXE_ANIMATOR_NAME} PRIVATE
  ${DALICORE_CFLAGS_OTHER}
  ${DALIADAPTOR_CFLAGS_OTHER}
  ${DALI_UI_FOUNDATION_CFLAGS_OTHER}
  ${DALI_UI_COMPONENTS_CFLAGS_OTHER}
  ${DALI_UI_TEST_APP_CFLAGS}
)

  ${DALICORE_CFLAGS_OTHER}
  ${DALIADAPTOR_CFLAGS_OTHER}
  ${DALI_UI_FOUNDATION_CFLAGS_OTHER}
  ${DALI_UI_COMPONENTS_CFLAGS_OTHER}
  ${DALI_UI_TEST_APP_CFLAGS}
)
TARGET_COMPILE_OPTIONS(${EXE_SUBTREE_NAME} PRIVATE
  ${DALICORE_CFLAGS_OTHER}
  ${DALIADAPTOR_CFLAGS_OTHER}
  ${DALI_UI_FOUNDATION_CFLAGS_OTHER}
  ${DALI_UI_COMPONENTS_CFLAGS_OTHER}
  ${DALI_UI_TEST_APP_CFLAGS}
)
TARGET_COMPILE_OPTIONS(${EXE_GRID_REORDER_NAME} PRIVATE
  ${DALICORE_CFLAGS_OTHER}
  ${DALIADAPTOR_CFLAGS_OTHER}
  ${DALI_UI_FOUNDATION_CFLAGS_OTHER}
  ${DALI_UI_COMPONENTS_CFLAGS_OTHER}
  ${DALI_UI_TEST_APP_CFLAGS}
)

# The grid reorder example loads SVG icons from its resource directory.
TARGET_COMPILE_DEFINITIONS(${EXE_GRID_REORDER_NAME} PRIVATE
  RESOURCES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/res/"
)

# Copy resources next to the built binaries.
file(COPY ${CMAKE_CURRENT_SOURCE_DIR}/res DESTINATION ${CMAKE_CURRENT_SOURCE_DIR}/bin)
