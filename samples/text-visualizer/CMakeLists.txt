CMAKE_MINIMUM_REQUIRED(VERSION 3.8.2)

PROJECT(text-visualizer.example)

SET(CMAKE_CXX_STANDARD 17)
SET(CMAKE_C_STANDARD 99)

# Find required packages
FIND_PACKAGE(PkgConfig REQUIRED)
PKG_CHECK_MODULES(DALICORE REQUIRED dali2-core)
PKG_CHECK_MODULES(DALIADAPTOR REQUIRED dali2-adaptor)
PKG_CHECK_MODULES(DALI_UI_FOUNDATION REQUIRED dali2-ui-foundation)
PKG_CHECK_MODULES(DALI_UI_COMPONENTS REQUIRED dali2-ui-components)

# Add executables
SET(EXE_TEXT_VISUALIZER_PERFORMANCE_NAME text-visualizer-performance.example)
ADD_EXECUTABLE(${EXE_TEXT_VISUALIZER_PERFORMANCE_NAME} text-visualizer-performance-example.cpp)

SET(EXE_TEXT_BREAKER_NAME text-breaker.example)
ADD_EXECUTABLE(${EXE_TEXT_BREAKER_NAME} text-breaker-example.cpp)

# Set output directory
SET_TARGET_PROPERTIES(${EXE_TEXT_VISUALIZER_PERFORMANCE_NAME} ${EXE_TEXT_BREAKER_NAME} PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")

INSTALL(TARGETS ${EXE_TEXT_VISUALIZER_PERFORMANCE_NAME} ${EXE_TEXT_BREAKER_NAME} RUNTIME DESTINATION)

# Link libraries
TARGET_LINK_LIBRARIES(${EXE_TEXT_VISUALIZER_PERFORMANCE_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)

TARGET_LINK_LIBRARIES(${EXE_TEXT_BREAKER_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)

# Add link directories
TARGET_LINK_DIRECTORIES(${EXE_TEXT_VISUALIZER_PERFORMANCE_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)

TARGET_LINK_DIRECTORIES(${EXE_TEXT_BREAKER_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)

# Include directories
TARGET_INCLUDE_DIRECTORIES(${EXE_TEXT_VISUALIZER_PERFORMANCE_NAME} PRIVATE
  ${DALICORE_INCLUDE_DIRS}
  ${DALIADAPTOR_INCLUDE_DIRS}
  ${DALI_UI_FOUNDATION_INCLUDE_DIRS}
  ${DALI_UI_COMPONENTS_INCLUDE_DIRS}
)

TARGET_INCLUDE_DIRECTORIES(${EXE_TEXT_BREAKER_NAME} PRIVATE
  ${DALICORE_INCLUDE_DIRS}
  ${DALIADAPTOR_INCLUDE_DIRS}
  ${DALI_UI_FOUNDATION_INCLUDE_DIRS}
  ${DALI_UI_COMPONENTS_INCLUDE_DIRS}
)

# Compiler flags
TARGET_COMPILE_OPTIONS(${EXE_TEXT_VISUALIZER_PERFORMANCE_NAME} PRIVATE
  ${DALICORE_CFLAGS_OTHER}
  ${DALIADAPTOR_CFLAGS_OTHER}
  ${DALI_UI_FOUNDATION_CFLAGS_OTHER}
  ${DALI_UI_COMPONENTS_CFLAGS_OTHER}
)

TARGET_COMPILE_OPTIONS(${EXE_TEXT_BREAKER_NAME} PRIVATE
  ${DALICORE_CFLAGS_OTHER}
  ${DALIADAPTOR_CFLAGS_OTHER}
  ${DALI_UI_FOUNDATION_CFLAGS_OTHER}
  ${DALI_UI_COMPONENTS_CFLAGS_OTHER}
)
