CMAKE_MINIMUM_REQUIRED(VERSION 3.8.2)

PROJECT(text.example)

SET(CMAKE_CXX_STANDARD 17)
SET(CMAKE_C_STANDARD 99)

# Find required packages
FIND_PACKAGE(PkgConfig REQUIRED)
PKG_CHECK_MODULES(DALICORE REQUIRED dali2-core)
PKG_CHECK_MODULES(DALIADAPTOR REQUIRED dali2-adaptor)
PKG_CHECK_MODULES(DALI_UI_FOUNDATION REQUIRED dali2-ui-foundation)
PKG_CHECK_MODULES(DALI_UI_COMPONENTS REQUIRED dali2-ui-components)

# Add executables
SET(EXE_NAME text.example)
ADD_EXECUTABLE(${EXE_NAME} text-example.cpp)

SET(EXE_LAYOUT_DIRECTION_NAME text-layout-direction.example)
ADD_EXECUTABLE(${EXE_LAYOUT_DIRECTION_NAME} text-layout-direction-example.cpp)

SET(EXE_MARKUP_NAME text-markup.example)
ADD_EXECUTABLE(${EXE_MARKUP_NAME} text-markup-example.cpp)

SET(EXE_MARQUEE_NAME text-marquee.example)
ADD_EXECUTABLE(${EXE_MARQUEE_NAME} text-marquee-example.cpp)

SET(EXE_STYLE_NAME text-style.example)
ADD_EXECUTABLE(${EXE_STYLE_NAME} text-style-example.cpp)

SET(EXE_STYLE_BEVEL_NAME text-style-bevel.example)
ADD_EXECUTABLE(${EXE_STYLE_BEVEL_NAME} text-style-bevel-example.cpp)

SET(EXE_TEXT_FIT_NAME text-fit.example)
ADD_EXECUTABLE(${EXE_TEXT_FIT_NAME} text-fit-example.cpp)

SET(EXE_TEXT_FIT_CANDIDATE_NAME text-fit-candidate.example)
ADD_EXECUTABLE(${EXE_TEXT_FIT_CANDIDATE_NAME} text-fit-candidate-example.cpp)

# Set output directory
SET_TARGET_PROPERTIES(${EXE_NAME} ${EXE_LAYOUT_DIRECTION_NAME} ${EXE_MARKUP_NAME} ${EXE_MARQUEE_NAME} ${EXE_STYLE_NAME} ${EXE_STYLE_BEVEL_NAME} ${EXE_TEXT_FIT_NAME} ${EXE_TEXT_FIT_CANDIDATE_NAME} PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")

INSTALL(TARGETS ${EXE_NAME} ${EXE_LAYOUT_DIRECTION_NAME} ${EXE_MARKUP_NAME} ${EXE_MARQUEE_NAME} ${EXE_STYLE_NAME} ${EXE_STYLE_BEVEL_NAME} ${EXE_TEXT_FIT_NAME} ${EXE_TEXT_FIT_CANDIDATE_NAME} RUNTIME DESTINATION)

# Link libraries
TARGET_LINK_LIBRARIES(${EXE_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)

TARGET_LINK_LIBRARIES(${EXE_LAYOUT_DIRECTION_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)

TARGET_LINK_LIBRARIES(${EXE_MARKUP_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)

TARGET_LINK_LIBRARIES(${EXE_MARQUEE_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)

TARGET_LINK_LIBRARIES(${EXE_STYLE_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)

TARGET_LINK_LIBRARIES(${EXE_STYLE_BEVEL_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)

TARGET_LINK_LIBRARIES(${EXE_TEXT_FIT_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)

TARGET_LINK_LIBRARIES(${EXE_TEXT_FIT_CANDIDATE_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${DALI_UI_COMPONENTS_LIBRARIES}
)

# Add link directories
TARGET_LINK_DIRECTORIES(${EXE_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)

TARGET_LINK_DIRECTORIES(${EXE_LAYOUT_DIRECTION_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)

TARGET_LINK_DIRECTORIES(${EXE_MARKUP_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)

TARGET_LINK_DIRECTORIES(${EXE_MARQUEE_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib

TARGET_LINK_DIRECTORIES(${EXE_TEXT_FIT_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)

TARGET_LINK_DIRECTORIES(${EXE_TEXT_FIT_CANDIDATE_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)
