CMAKE_MINIMUM_REQUIRED(VERSION 3.8.2)

PROJECT(dali-ui-widget-app)

SET(CMAKE_CXX_STANDARD 17)
SET(CMAKE_C_STANDARD 99)

# This provider uses WidgetApplication and Tizen platform IPC libraries.
IF(NOT TIZEN)
  RETURN()
ENDIF()

IF(NOT DALI_API_MAJOR)
  SET(DALI_API_MAJOR 3)
ENDIF()

FIND_PACKAGE(PkgConfig REQUIRED)
PKG_CHECK_MODULES(DALICORE           REQUIRED dali${DALI_API_MAJOR}-core)
PKG_CHECK_MODULES(DALIADAPTOR        REQUIRED dali${DALI_API_MAJOR}-adaptor)
PKG_CHECK_MODULES(DALI_UI_FOUNDATION REQUIRED dali${DALI_API_MAJOR}-ui-foundation)
PKG_CHECK_MODULES(BUNDLE             REQUIRED bundle)
PKG_CHECK_MODULES(DLOG               REQUIRED dlog)
PKG_CHECK_MODULES(MESSAGE_PORT       REQUIRED message-port)

SET(EXE_NAME widget-app)

ADD_EXECUTABLE(${EXE_NAME} widget-app.cpp)

SET_TARGET_PROPERTIES(${EXE_NAME} PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin"
  POSITION_INDEPENDENT_CODE TRUE)

# Tizen launchpad loads the app binary via dlopen(), which requires
# a PIE executable (ET_DYN). Without -pie here the binary comes out
# as ET_EXEC and launchpad fails with "cannot dynamically load executable".
TARGET_LINK_OPTIONS(${EXE_NAME} PRIVATE -pie -Wl,--export-dynamic)

INSTALL(TARGETS ${EXE_NAME} RUNTIME DESTINATION bin)

TARGET_LINK_LIBRARIES(${EXE_NAME}
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
  ${BUNDLE_LIBRARIES}
  ${DLOG_LIBRARIES}
  ${MESSAGE_PORT_LIBRARIES}
)

TARGET_LINK_DIRECTORIES(${EXE_NAME} PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)

TARGET_INCLUDE_DIRECTORIES(${EXE_NAME} PRIVATE
  ${DALICORE_INCLUDE_DIRS}
  ${DALIADAPTOR_INCLUDE_DIRS}
  ${DALI_UI_FOUNDATION_INCLUDE_DIRS}
  ${BUNDLE_INCLUDE_DIRS}
  ${DLOG_INCLUDE_DIRS}
  ${MESSAGE_PORT_INCLUDE_DIRS}
)

# Match dali-demo: hide all symbols by default, but keep DALI_EXPORT_API
# (applied to main()) exported so the Tizen installer's EXEC_CHECKER
# can locate the main symbol. HIDE_DALI_INTERNALS activates the
# visibility("default") form of DALI_EXPORT_API in dali-core headers.
TARGET_COMPILE_DEFINITIONS(${EXE_NAME} PRIVATE HIDE_DALI_INTERNALS)

TARGET_COMPILE_OPTIONS(${EXE_NAME} PRIVATE
  ${DALICORE_CFLAGS_OTHER}
  ${DALIADAPTOR_CFLAGS_OTHER}
  ${DALI_UI_FOUNDATION_CFLAGS_OTHER}
  ${BUNDLE_CFLAGS_OTHER}
  ${DLOG_CFLAGS_OTHER}
  -fvisibility=hidden
)
