CMAKE_MINIMUM_REQUIRED(VERSION 3.8.2)

PROJECT(hello-world-example)

SET(CMAKE_CXX_STANDARD 17)
SET(CMAKE_C_STANDARD 99)

# Find required packages
FIND_PACKAGE(PkgConfig REQUIRED)
PKG_CHECK_MODULES(DALICORE REQUIRED dali2-core)
PKG_CHECK_MODULES(DALIADAPTOR REQUIRED dali2-adaptor)
PKG_CHECK_MODULES(DALITOOLKIT REQUIRED dali2-toolkit)
PKG_CHECK_MODULES(DALI_UI_FOUNDATION REQUIRED dali2-ui-foundation)

# Add executable
ADD_EXECUTABLE(hello-world-example hello-world-example.cpp)

# Link libraries
TARGET_LINK_LIBRARIES(hello-world-example
  ${DALICORE_LIBRARIES}
  ${DALIADAPTOR_LIBRARIES}
  ${DALITOOLKIT_LIBRARIES}
  ${DALI_UI_FOUNDATION_LIBRARIES}
)

# Add link directories
TARGET_LINK_DIRECTORIES(hello-world-example PRIVATE
  $ENV{DESKTOP_PREFIX}/lib
)

# Include directories
TARGET_INCLUDE_DIRECTORIES(hello-world-example PRIVATE
  ${DALICORE_INCLUDE_DIRS}
  ${DALIADAPTOR_INCLUDE_DIRS}
  ${DALITOOLKIT_INCLUDE_DIRS}
  ${DALI_UI_FOUNDATION_INCLUDE_DIRS}
)

# Compiler flags
TARGET_COMPILE_OPTIONS(hello-world-example PRIVATE
  ${DALICORE_CFLAGS_OTHER}
  ${DALIADAPTOR_CFLAGS_OTHER}
  ${DALITOOLKIT_CFLAGS_OTHER}
  ${DALI_UI_FOUNDATION_CFLAGS_OTHER}
)
