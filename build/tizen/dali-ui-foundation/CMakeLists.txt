PROJECT(dali-ui-foundation)
SET(PROJECT_LIB_NAME dali2-ui-foundation)
SET(FOUNDATION_ROOT_SRC_DIR ${ROOT_SRC_DIR}/dali-ui-foundation)

# ---------------------------------------------------------
# 1. Shader Generator Configuration
# ---------------------------------------------------------
SET(SHADER_GENERATOR_NAME dali-ui-shader-generator)
SET(SHADER_GENERATOR_SOURCES ${FOUNDATION_ROOT_SRC_DIR}/shader-generator/shader-generator.cpp)

# Build shader generator executable for host
ADD_EXECUTABLE(${SHADER_GENERATOR_NAME} ${SHADER_GENERATOR_SOURCES})

# Set shader source and generated cpp file paths
SET(SHADER_SOURCE_DIR "${FOUNDATION_ROOT_SRC_DIR}/internal/graphics/shaders")
SET(SHADER_GENERATED_DIR "${FOUNDATION_ROOT_SRC_DIR}/internal/graphics/generated")
SET(BUILT_IN_SHADER_GEN_CPP "${SHADER_GENERATED_DIR}/builtin-shader-gen.cpp")

# Clean-up generated directory when 'make clean'
SET_DIRECTORY_PROPERTIES(PROPERTIES ADDITIONAL_MAKE_CLEAN_FILES "${SHADER_GENERATED_DIR}")

FILE(GLOB SHADERS_SRC "${SHADER_SOURCE_DIR}/*.vert" "${SHADER_SOURCE_DIR}/*.frag" )

# Custom command: Automatically generate builtin-shader-gen.cpp at build time
ADD_CUSTOM_COMMAND(OUTPUT ${BUILT_IN_SHADER_GEN_CPP}
                   DEPENDS ${SHADER_GENERATOR_NAME} ${SHADERS_SRC}
                   COMMAND ${SHADER_GENERATOR_NAME} ${SHADER_SOURCE_DIR} ${SHADER_GENERATED_DIR})

# Custom target to ensure dependency in parallel builds
ADD_CUSTOM_TARGET(generate_builtin_shaders DEPENDS ${BUILT_IN_SHADER_GEN_CPP})


# ---------------------------------------------------------
# 2. Collect and filter source files
# ---------------------------------------------------------
FILE(GLOB_RECURSE ALL_SOURCES "${FOUNDATION_ROOT_SRC_DIR}/*.cpp")

SET(SOURCES "")
FOREACH(SRC ${ALL_SOURCES})
  # Exclude shader-generator.cpp and the generated cpp from initial glob to avoid duplication
  IF(NOT SRC MATCHES "shader-generator.cpp" AND NOT SRC MATCHES "builtin-shader-gen.cpp")
    LIST(APPEND SOURCES ${SRC})
  ENDIF()
ENDFOREACH()

# Add code generated by shader generator to library sources
LIST(APPEND SOURCES ${BUILT_IN_SHADER_GEN_CPP})

# Add code generated for public API InvokeMethod wrappers
OPTION(ENABLE_INVOKE_METHOD_WRAPPERS "Generate InvokeMethod wrappers for exported public API handle methods" ON)

IF(ENABLE_INVOKE_METHOD_WRAPPERS)
  FIND_PACKAGE(Python3 COMPONENTS Interpreter REQUIRED)
  FIND_PROGRAM(INVOKE_METHOD_GENERATOR
               NAMES dali-invoke-method-generator dali-invoke-method-generator.py
               PATHS ${BIN_DIR} $ENV{DESKTOP_PREFIX}/bin ${ROOT_SRC_DIR}/../dali-core/scripts)

  IF(NOT INVOKE_METHOD_GENERATOR)
    MESSAGE(FATAL_ERROR "dali-invoke-method-generator not found. Build and install dali-core first.")
  ENDIF()

  SET(INVOKE_METHOD_GENERATED_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated/invoke-method)
  SET(INVOKE_METHOD_PUBLIC_API_CPP ${INVOKE_METHOD_GENERATED_DIR}/public-api-invoke-methods.cpp)
  SET(INVOKE_METHOD_CORE_PUBLIC_API_SOURCE_DIR ${FOUNDATION_ROOT_SRC_DIR}/../../dali-core/dali/public-api)
  SET(INVOKE_METHOD_ADAPTOR_PUBLIC_API_SOURCE_DIR ${FOUNDATION_ROOT_SRC_DIR}/../../dali-adaptor/dali/public-api)
  SET(INVOKE_METHOD_DALI_PUBLIC_API_INSTALL_DIR ${INCLUDE_DIR}/dali/public-api)
  IF(EXISTS ${INVOKE_METHOD_CORE_PUBLIC_API_SOURCE_DIR})
    SET(INVOKE_METHOD_CORE_PUBLIC_API_DIR ${INVOKE_METHOD_CORE_PUBLIC_API_SOURCE_DIR})
  ELSEIF(EXISTS ${INVOKE_METHOD_DALI_PUBLIC_API_INSTALL_DIR})
    SET(INVOKE_METHOD_CORE_PUBLIC_API_DIR ${INVOKE_METHOD_DALI_PUBLIC_API_INSTALL_DIR})
  ELSE()
    MESSAGE(FATAL_ERROR "dali-core public API headers not found. Build and install dali-core first.")
  ENDIF()
  IF(EXISTS ${INVOKE_METHOD_ADAPTOR_PUBLIC_API_SOURCE_DIR})
    SET(INVOKE_METHOD_ADAPTOR_PUBLIC_API_DIR ${INVOKE_METHOD_ADAPTOR_PUBLIC_API_SOURCE_DIR})
  ELSEIF(EXISTS ${INVOKE_METHOD_DALI_PUBLIC_API_INSTALL_DIR})
    SET(INVOKE_METHOD_ADAPTOR_PUBLIC_API_DIR ${INVOKE_METHOD_DALI_PUBLIC_API_INSTALL_DIR})
  ELSE()
    MESSAGE(FATAL_ERROR "dali-adaptor public API headers not found. Build and install dali-adaptor first.")
  ENDIF()
  FILE(GLOB_RECURSE INVOKE_METHOD_PUBLIC_API_HEADERS CONFIGURE_DEPENDS ${FOUNDATION_ROOT_SRC_DIR}/public-api/*.h)
  FILE(GLOB_RECURSE INVOKE_METHOD_CORE_PUBLIC_API_HEADERS CONFIGURE_DEPENDS ${INVOKE_METHOD_CORE_PUBLIC_API_DIR}/*.h)
  FILE(GLOB_RECURSE INVOKE_METHOD_ADAPTOR_PUBLIC_API_HEADERS CONFIGURE_DEPENDS ${INVOKE_METHOD_ADAPTOR_PUBLIC_API_DIR}/*.h)
  FILE(GLOB_RECURSE INVOKE_METHOD_REGISTRATION_SCAN_FILES CONFIGURE_DEPENDS
       ${FOUNDATION_ROOT_SRC_DIR}/*.cpp
       ${FOUNDATION_ROOT_SRC_DIR}/*.h
       ${FOUNDATION_ROOT_SRC_DIR}/*.hpp)

  ADD_CUSTOM_COMMAND(OUTPUT ${INVOKE_METHOD_PUBLIC_API_CPP}
                     COMMAND ${Python3_EXECUTABLE} ${INVOKE_METHOD_GENERATOR}
                             --scan-root ${FOUNDATION_ROOT_SRC_DIR}/public-api
                             --include-prefix dali-ui-foundation/public-api
                             --dependency-scan-root ${INVOKE_METHOD_CORE_PUBLIC_API_DIR}
                             --dependency-include-prefix dali/public-api
                             --dependency-scan-root ${INVOKE_METHOD_ADAPTOR_PUBLIC_API_DIR}
                             --dependency-include-prefix dali/public-api
                             --dependency-export-api DALI_CORE_API
                             --dependency-export-api DALI_ADAPTOR_API
                             --registration-scan-root ${FOUNDATION_ROOT_SRC_DIR}
                             --namespace Dali::Ui
                             --auto-export-api DALI_UI_API
                             --allow-type Duration
                             --allow-type Dali::Ui::Duration
                             --allow-type UiColor
                             --allow-type Dali::Ui::UiColor
                             --output ${INVOKE_METHOD_PUBLIC_API_CPP}
                     DEPENDS ${INVOKE_METHOD_GENERATOR}
                             ${INVOKE_METHOD_PUBLIC_API_HEADERS}
                             ${INVOKE_METHOD_CORE_PUBLIC_API_HEADERS}
                             ${INVOKE_METHOD_ADAPTOR_PUBLIC_API_HEADERS}
                             ${INVOKE_METHOD_REGISTRATION_SCAN_FILES}
                     COMMENT "Generating DALi invoke-method wrappers for exported UI public API"
                     VERBATIM)

  LIST(APPEND SOURCES ${INVOKE_METHOD_PUBLIC_API_CPP})
ENDIF()

# ---------------------------------------------------------
# 3. Create shared library
# ---------------------------------------------------------
ADD_LIBRARY(${PROJECT_LIB_NAME} SHARED ${SOURCES})

IF(WIN32)
  TARGET_COMPILE_DEFINITIONS(${PROJECT_LIB_NAME} PRIVATE BUILDING_DALI_UI_FOUNDATION)
ENDIF()

# Ensure shaders are generated before the library starts compiling its sources
ADD_DEPENDENCIES(${PROJECT_LIB_NAME} generate_builtin_shaders)

# Run animation autogen script
IF (NOT GBS_BUILD)
  DALI_UI_AUTOGEN_ANIMATION_SPEC(${PROJECT_LIB_NAME} "${FOUNDATION_ROOT_SRC_DIR}/public-api")
ENDIF()


# ---------------------------------------------------------
# 4. Include directory configuration (add third-party and internal APIs, remove toolkit)
# ---------------------------------------------------------
TARGET_INCLUDE_DIRECTORIES(${PROJECT_LIB_NAME}
  PUBLIC
    $<BUILD_INTERFACE:${ROOT_SRC_DIR}>
    $<INSTALL_INTERFACE:include>
  PRIVATE
    ${FOUNDATION_ROOT_SRC_DIR}/third-party
    ${FOUNDATION_ROOT_SRC_DIR}/internal
    ${FOUNDATION_ROOT_SRC_DIR}/integration-api
    ${DALICORE_INCLUDE_DIRS}
    ${DALIADAPTOR_INCLUDE_DIRS}
    ${DALI_UI_GETTEXT_INCLUDE_DIRS}
    ${VCPKG_INCLUDE_DIR}
    ${INCLUDE_DIR}
)


# ---------------------------------------------------------
# 5. Link dependencies (remove toolkit, add GLES)
# ---------------------------------------------------------
# Note: PKG_CHECK_MODULES(OPENGLES glesv2 egl) must be declared in the top-level CMakeLists.txt
TARGET_LINK_LIBRARIES(${PROJECT_LIB_NAME}
  PUBLIC ${DALI_UI_PLATFORM_LIBRARIES}
  PRIVATE ${DALI_UI_GETTEXT_LIBRARIES} ${COVERAGE}
)

SET_TARGET_PROPERTIES(${PROJECT_LIB_NAME}
  PROPERTIES
  VERSION ${DALI_UI_VERSION}
  SOVERSION ${DALI_UI_VERSION_MAJOR}
  CLEAN_DIRECT_OUTPUT 1
)


# ---------------------------------------------------------
# 6. Install configuration (add integration-api and shader generator)
# ---------------------------------------------------------
# Install library and CMake package metadata.
IF(INSTALL_CMAKE_MODULES)
  INSTALL(TARGETS ${PROJECT_LIB_NAME}
    EXPORT ${PROJECT_LIB_NAME}-targets
    LIBRARY DESTINATION ${DALI_UI_INSTALL_LIB_DIR}
    ARCHIVE DESTINATION ${DALI_UI_INSTALL_LIB_DIR}
    RUNTIME DESTINATION ${DALI_UI_INSTALL_BIN_DIR}
  )

  INSTALL(EXPORT ${PROJECT_LIB_NAME}-targets
    NAMESPACE ${PROJECT_LIB_NAME}::
    FILE ${PROJECT_LIB_NAME}-targets.cmake
    DESTINATION share/${PROJECT_LIB_NAME}
  )

  IF(WIN32)
    SET(PACKAGE_DEPENDENCIES
"    find_dependency(dali2-core CONFIG REQUIRED)\n    find_dependency(dali2-adaptor CONFIG REQUIRED)\n    find_dependency(unofficial-angle CONFIG REQUIRED)\n")
  ELSE()
    SET(PACKAGE_DEPENDENCIES
"    find_dependency(dali2-core CONFIG REQUIRED)\n    find_dependency(dali2-adaptor CONFIG REQUIRED)\n")
  ENDIF()

  FILE(WRITE ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_LIB_NAME}-config.cmake
"include(CMakeFindDependencyMacro)\n${PACKAGE_DEPENDENCIES}include(\${CMAKE_CURRENT_LIST_DIR}/${PROJECT_LIB_NAME}-targets.cmake)\n")
  INSTALL(FILES ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_LIB_NAME}-config.cmake
          DESTINATION share/${PROJECT_LIB_NAME})
ELSE()
  INSTALL(TARGETS ${PROJECT_LIB_NAME}
    LIBRARY DESTINATION ${DALI_UI_INSTALL_LIB_DIR}
    ARCHIVE DESTINATION ${DALI_UI_INSTALL_LIB_DIR}
    RUNTIME DESTINATION ${DALI_UI_INSTALL_BIN_DIR}
  )
ENDIF()

# Install headers
SET(PROJECT_INCLUDE_DIR ${INCLUDE_DIR}/${PROJECT_NAME})
INSTALL(DIRECTORY ${FOUNDATION_ROOT_SRC_DIR}/public-api/
        DESTINATION ${PROJECT_INCLUDE_DIR}/public-api
        FILES_MATCHING PATTERN "*.h")
INSTALL(DIRECTORY ${FOUNDATION_ROOT_SRC_DIR}/extension-api/
        DESTINATION ${PROJECT_INCLUDE_DIR}/extension-api
        FILES_MATCHING PATTERN "*.h")
INSTALL(DIRECTORY ${FOUNDATION_ROOT_SRC_DIR}/integration-api/
        DESTINATION ${PROJECT_INCLUDE_DIR}/integration-api
        FILES_MATCHING PATTERN "*.h")

# Install dali-ui-foundation.h
INSTALL( FILES ${FOUNDATION_ROOT_SRC_DIR}/dali-ui-foundation.h DESTINATION ${PROJECT_INCLUDE_DIR} )
INSTALL( FILES ${FOUNDATION_ROOT_SRC_DIR}/dali-ui-foundation-extension.h DESTINATION ${PROJECT_INCLUDE_DIR} )
INSTALL( FILES ${FOUNDATION_ROOT_SRC_DIR}/dali-ui-foundation-integ.h DESTINATION ${PROJECT_INCLUDE_DIR} )

# Install shader generator executable (for development)
INSTALL(TARGETS ${SHADER_GENERATOR_NAME} RUNTIME DESTINATION ${DALI_UI_INSTALL_BIN_DIR})

# Install pkg config
IF( ENABLE_PKG_CONFIGURE )
  SET( DEV_INCLUDE_PATH ${INCLUDE_DIR} )
  SET( PKG_CFG_FILE ${CMAKE_CURRENT_SOURCE_DIR}/${PROJECT_LIB_NAME}.pc )
  SET( INTEGRATION_PKG_CFG_FILE ${CMAKE_CURRENT_SOURCE_DIR}/${PROJECT_LIB_NAME}-integration.pc )
  CONFIGURE_FILE( ${PKG_CFG_FILE}.in ${PKG_CFG_FILE} @ONLY )
  CONFIGURE_FILE( ${INTEGRATION_PKG_CFG_FILE}.in ${INTEGRATION_PKG_CFG_FILE} @ONLY )
  INSTALL( FILES ${PKG_CFG_FILE} DESTINATION ${LIB_DIR}/pkgconfig )
  INSTALL( FILES ${INTEGRATION_PKG_CFG_FILE} DESTINATION ${LIB_DIR}/pkgconfig )
ENDIF()

# Install resources
INSTALL(DIRECTORY ${FOUNDATION_ROOT_SRC_DIR}/images/
        DESTINATION ${dataReadOnlyInstallDir}/ui/images)
